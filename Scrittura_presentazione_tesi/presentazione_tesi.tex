\begin{frame}{3. Fit spettro energia}
\begin{columns}
    \begin{column}{0.55\textwidth}
        \begin{block}{Fluenza differenziale}
            \vspace{-0.5em}
            \begin{equation*}
               \hspace{-2em} F(E) = \int_{t_1}^{t_2} \Phi(E)\,dt
               \qquad \small t_1,\,t_2 = \text{durata evento}
            \end{equation*}
            \vspace{-1.0em}
        \end{block}
        \vspace{0.2em}
        \begin{block}{Modello spettrale: Band function}
            \small
            \vspace{-1.5em}
            \begin{equation*}
                F_{\text{Band}}(E) = 
                \begin{cases}
                    \textcolor{teipel_red!70}{A}\,E^{-\textcolor{teipel_red!70}{\gamma_a}}\,
                    e^{-E/\textcolor{teipel_red!70}{E_0}} 
                    \\[0.3em]
                    \textcolor{teipel_red!70}{A}\,E^{-\textcolor{teipel_red!70}{\gamma_b}}\,
                    [(\textcolor{teipel_red!70}{\gamma_b}\!-\!\textcolor{teipel_red!70}{\gamma_a})\textcolor{teipel_red!70}{E_0}]^{(\textcolor{teipel_red!70}{\gamma_b}-\textcolor{teipel_red!70}{\gamma_a})}\,
                    e^{(\textcolor{teipel_red!70}{\gamma_a}-\textcolor{teipel_red!70}{\gamma_b})}
                \end{cases}
            \end{equation*}
        \vspace{-1.0em}
        \end{block}
        \vspace{0.2em}
        \begin{block}{Incertezze}
            \small
            \begin{flushleft}
            \vspace{0.2em}
            $\sigma_{\text{tot}} = \sqrt{\sigma_{\text{sys}}^2 + \sigma_{\text{stat}}^2}
            \;\Longrightarrow$ Montecarlo parametrico\\[0.3em]
            \hspace*{0pt}$\mathcal{N}(\hat{\mathbf{p}},\, \mathbf{C}_{\text{stab}})
            \;\Longrightarrow\;
            \textcolor{violet!30}{F_{\text{med}}(E)} \;\pm\;\textcolor{violet!80}{[F_{16},\, F_{84}]}$\; 68\% CL
            \vspace{-0.3em}
            \end{flushleft}
        \end{block}
    \end{column}
    
    \begin{column}{0.43\textwidth}
        \centering
        \includegraphics[width=\textwidth]{Images/JAN2005_spectral_fit_antra.pdf}\\[0.2em]
        {\tiny Esempio: fit JAN 2005}
    \end{column}
\end{columns}

\begin{frame}{Calcolatore SR-NIEL}
    \begin{columns}[T]
        \column{0.60\textwidth}
        \begin{itemize}
            \item Stopping power elettronico: $S(E) = -\frac{dE}{dx}$
            \item Bethe-Bloch fino a 5GeV/amu, correzioni SRIM a basse energie\\
            $-\frac{dE}{dx} = \frac{0.1535 \, \rho \, z^2 Z}{A \, \beta^2} \left[ \ln\!\left(\frac{2m v^2 W_m}{I^2(1-\beta^2)}\right) - 2\beta^2 - \delta - U \right]$
            \item \textit{Continuously Slowing Down Approximation} (CSDA) per propagare spettro
            \item Energia depositata: costante e completamente assorbita dal materiale
            \item No particelle secondarie, no raggi $\delta$
            \item $\Phi_{\text{plan}}(E) =\Phi_{\text{dir}}(E) \int_{2\pi} \cos(\theta) \, d\Omega = \textcolor{teipel_red!80}{\pi} \, \Phi_{\mathrm{dir}}$
            \item $\Phi_{\mathrm{sfer}}(E) = \Phi_{\mathrm{dir}}(E) \int_{4\pi} d\Omega = \textcolor{teipel_red!80}{4\pi} \, \Phi_{\mathrm{dir}}(E)$
             
        \end{itemize}

\begin{frame}{Risultati: funzione di trasferimento dose--fluenza}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{itemize}
            \item La TID dipende dall'intera forma dello spettro
            \item Spettro: $ \Phi(E) = A E^{-\gamma} $
            \item Fluenza: $ J(>E_0) = \int_{E_0}^{\infty} A E^{-\gamma} \, dE = \frac{AE_0^{1-\gamma}}{\gamma-1}$
            \item Dose: $D = \int A E^{-\gamma}\cdot S(E) \, dE$
            \item Entrambe $\propto A$ $\Rightarrow$ $D \propto J(>E_0)^{b}$
            \item Individuando la soglia energetica ottimale è possibile condensare le differenze spettrali in una relazione lineare $R^2 = 0.998$, permettendo di stimare la TID con un errore dell'ordine del 10\%
            \vspace{5pt}
            $$
                \log(D) = a + b \cdot \log(J(>E_0)) \qquad \sigma_{\log D} = \frac{\Delta D}{D\cdot \ln(10)}
            $$
            
            
             
        \end{itemize}
