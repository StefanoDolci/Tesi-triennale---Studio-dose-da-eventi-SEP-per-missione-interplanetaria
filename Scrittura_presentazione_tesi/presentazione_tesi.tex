\begin{frame}{Dataset eventi SEP storici}
    \begin{columns}

            \begin{itemize}
                \item 8 eventi dal 1972--2024: 
                \begin{itemize}                    \item AUG 1972, OCT 1989 
                    \item OCT 2003, JAN 2005
                    \item JUL 2012, SEP 2017
                    \item OCT 2021, MAY 2024
                \end{itemize}
                \item Strumenti: GOES, ACE, IMP-8, STEREO-A
                \item Preprocessing: $\Delta t = 30$~min.
            \end{itemize}
			\vspace{0.3em}
                \begin{block}{Soglia e sottrazione background}
                \vspace{-1.5em}
                \begin{flalign*}
                    \Phi_{\text{thr}} = \mu + 3\sigma &&
                \end{flalign*}
                \vspace{-1.5em}
                \begin{flalign*}
                    \Phi_{\text{net}}(t) = 
                    \begin{cases}
                        \Phi_{\text{raw}}(t) - \mu & \text{se } \Phi_{\text{raw}}(t) \geq \Phi_{\text{thr}} \\
                        0 \; (\text{NaN}) & \text{altrimenti}
                    \end{cases} &&
                \end{flalign*}
                \end{block}
        \end{column}

\begin{frame}{Fit spettro}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \begin{block}{Fluenza differenziale}
                \vspace{-0.3em}
                \begin{equation*}
                    F(E) = \int_{t_1}^{t_2} \Phi(E) \, dt
                \end{equation*}
                \vspace{-0.8em}
            \end{block}
            \vspace{0.2em}
            \begin{block}{Modelli spettrali}
                \vspace{-0.2em}
                {\small \textbf{Ellison-Ramaty}:}
                \vspace{-0.5em}
                \begin{equation*}
                    \Phi_{E\text{-}R}(E) = A \, E^{-\gamma} \, e^{-E/E_r}
                \end{equation*}
                \vspace{-0.8em}
                {\small \textbf{Band function}:}
                \vspace{-0.3em}
                \begin{gather*}
                    \small
                    \Phi_{\text{Band}}(E) = 
                    \begin{cases}
                        A E^{-\gamma_a} e^{-E/E_0} \\[4pt]
                        A E^{-\gamma_b} [(\gamma_b\!-\!\gamma_a)E_0]^{(\gamma_b-\gamma_a)} e^{(\gamma_a-\gamma_b)}
                    \end{cases}
                \end{gather*}
                \vspace{-1em}
            \end{block}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=\textwidth,keepaspectratio]{JAN2005_spectral_fit.pdf}
            \\[2pt]
            {\tiny Esempio: fit per lo spettro di gennaio 2005}
        \end{column}
    \end{columns}
\end{frame}
