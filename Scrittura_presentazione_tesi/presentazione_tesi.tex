\begin{frame}{3. Fit spettro}
\begin{columns}
    \begin{column}{0.55\textwidth}
        \begin{block}{Fluenza differenziale}
            \vspace{-0.5em}
            \begin{equation*}
                F(E) = \int_{t_1}^{t_2} \Phi(E)\,dt
            \end{equation*}
            \vspace{-1.0em}
        \end{block}
        \vspace{0.2em}
        \begin{block}{Modello spettrale: Band function}
            \small
            \vspace{-0.8em}
            \begin{equation*}
                \Phi_{\text{Band}}(E) = 
                \begin{cases}
                    \textcolor{teipel_red}{A}\,E^{-\textcolor{teipel_red}{\gamma_a}}\,
                    e^{-E/\textcolor{teipel_red}{E_0}} 
                    \\[0.3em]
                    \textcolor{teipel_red}{A}\,E^{-\textcolor{teipel_red}{\gamma_b}}\,
                    [(\textcolor{teipel_red}{\gamma_b}\!-\!\textcolor{teipel_red}{\gamma_a})\textcolor{teipel_red}{E_0}]^{(\textcolor{teipel_red}{\gamma_b}-\textcolor{teipel_red}{\gamma_a})}\,
                    e^{(\textcolor{teipel_red}{\gamma_a}-\textcolor{teipel_red}{\gamma_b})}
                \end{cases}
            \end{equation*}
        \end{block}
        {\tiny 4 parametri liberi: \textcolor{teipel_red}{$A,\;\gamma_a,\;\gamma_b,\;E_0$} \quad|\quad Incertezze: MC parametrico, CL 68\% }
    \end{column}
    
    \begin{column}{0.42\textwidth}
        \centering
        \includegraphics[width=\textwidth]{Images/JAN2005_spectral_fit_dark1.pdf}\\[0.2em]
        {\tiny Esempio: fit JAN 2005}
    \end{column}
\end{columns}
\end{frame}
